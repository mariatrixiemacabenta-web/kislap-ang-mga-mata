<script>

const music = document.getElementById("backgroundMusic");
const clickSound = document.getElementById("clickSound");

function playClick() {
    clickSound.currentTime = 0;
    clickSound.play().catch(() => {});
}

function toggleMusic() {
    playClick();

    if (music.paused) {
        music.play().then(() => {
            document.getElementById("musicStatus").innerText =
            "♫ Music is playing.";
        }).catch(() => {
            document.getElementById("musicStatus").innerText =
            "Please add a valid music.mp3 file.";
        });
    } else {
        music.pause();
        document.getElementById("musicStatus").innerText =
        "Music is paused.";
    }
}

function submitThought() {
    playClick();

    const name = document.getElementById("nickname").value.trim();
    const category = document.getElementById("category").value;
    const thought = document.getElementById("thought").value.trim();

    if (thought === "") {
        alert("Please write something before sharing! ♡");
        return;
    }

    const message = document.createElement("div");
    message.className = "message";

    const title = document.createElement("h3");
    title.textContent = category;

    const content = document.createElement("p");
    content.textContent = thought;

    const author = document.createElement("small");
    author.textContent = "— " + (name || "Anonymous");

    message.append(title, content, author);

    const container = document.getElementById("messages");

    if (container.querySelector("p")) {
        container.innerHTML = "";
    }

    container.prepend(message);

    document.getElementById("nickname").value = "";
    document.getElementById("thought").value = "";

    alert("Your feelings have been added! ♡");
}

</script>
